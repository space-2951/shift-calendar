function buildShiftText(){ const year = parseInt(document.getElementById('yearInput').value, 10); const month = parseInt(document.getElementById('monthInput').value, 10); const dayHeaders = ['日','月','火','水','木','金','土']; const table = document.querySelector('#card table'); const cells = Array.from(table.querySelectorAll('tbody td')).filter(td => !td.classList.contains('empty')); const holidays = getHolidays(year); const daysInMonth = new Date(year, month, 0).getDate(); const dayLines = []; let attendCount = 0; for (let day = 1; day <= daysInMonth; day++){ const d = new Date(year, month-1, day); const dow = dayHeaders[d.getDay()]; const key = `${year}-${pad(month)}-${pad(day)}`; const holidayName = holidays[key]; const holidaySuffix = holidayName ? `祝:${holidayName}` : ''; const td = cells.find(c => { const daynumEl = c.querySelector('.daynum'); return daynumEl && parseInt(daynumEl.textContent, 10) === day; }); const isAttend = td ? td.classList.contains('attend') : false; if (isAttend) { attendCount++; dayLines.push(`【${month}/${day}(${dow})${holidaySuffix} 】🟩`); } // 通所日じゃない日はここで何もしない(行を作らない) } const noteEl = document.getElementById('noteInput'); const noteText = noteEl ? noteEl.value.trim() : ''; let text = `【シフト】\n${year}年${month}月\n\n`; text += dayLines.join('\n'); text += `\n\n通所日 全${attendCount}日`; if (noteText) { text += `\n\n【メモ】\n${noteText}`; } return text; }
